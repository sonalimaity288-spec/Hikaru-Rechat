    local result = {}

    for _, id in ipairs(ids) do
        table.insert(result, {
            Name = "Sticker",
            Id = "rbxassetid://" .. tostring(id)
        })
    end

    return result
end

local function makeCorner(object, radius)
    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, radius)
    corner.Parent = object
end

local function makeStroke(object, color, thickness)
    local stroke = Instance.new("UIStroke")
    stroke.Color = color
    stroke.Thickness = thickness or 1
    stroke.Transparency = 0.25
    stroke.Parent = object
    return stroke
end

local function makeButton(parent, text, size, position, color)
    local button = Instance.new("TextButton")
    button.Size = size
    button.Position = position
    button.BackgroundColor3 = color
    button.TextColor3 = Theme.Text
    button.Text = text
    button.TextScaled = true
    button.Font = Enum.Font.GothamBold
    button.AutoButtonColor = true
    button.BorderSizePixel = 0
    button.Parent = parent

    makeCorner(button, 12)

    return button
end

local function makeLabel(parent, text, size, position)
    local label = Instance.new("TextLabel")
    label.Size = size
    label.Position = position
    label.BackgroundTransparency = 1
    label.Text = text
    label.TextColor3 = Theme.Text
    label.TextScaled = true
    label.Font = Enum.Font.GothamBold
    label.Parent = parent
    return label
end

local function showStickers()
    clearPage()

    local categoryBar = Instance.new("ScrollingFrame")
    categoryBar.Name = "Categories"
    categoryBar.Size = UDim2.new(1, -12, 0, 42)
    categoryBar.Position = UDim2.new(0, 6, 0, 5)
    categoryBar.BackgroundTransparency = 1
    categoryBar.BorderSizePixel = 0
    categoryBar.ScrollBarThickness = 2
    categoryBar.ScrollingDirection = Enum.ScrollingDirection.X
    categoryBar.AutomaticCanvasSize = Enum.AutomaticSize.X
    categoryBar.CanvasSize = UDim2.new(0, 0, 0, 0)
    categoryBar.Parent = currentPage

    local categoryLayout = Instance.new("UIListLayout")
    categoryLayout.FillDirection = Enum.FillDirection.Horizontal
    categoryLayout.Padding = UDim.new(0, 5)
    categoryLayout.SortOrder = Enum.SortOrder.LayoutOrder
    categoryLayout.Parent = categoryBar

    local gallery = Instance.new("ScrollingFrame")
    gallery.Name = "StickerGallery"
    gallery.Size = UDim2.new(1, -12, 1, -100)
    gallery.Position = UDim2.new(0, 6, 0, 52)
    gallery.BackgroundColor3 = Theme.Panel
    gallery.BorderSizePixel = 0
    gallery.ScrollBarThickness = 4
    gallery.AutomaticCanvasSize = Enum.AutomaticSize.Y
    gallery.CanvasSize = UDim2.new(0, 0, 0, 0)
    gallery.Parent = currentPage

    makeCorner(gallery, 12)

    local grid = Instance.new("UIGridLayout")
    grid.CellSize = UDim2.new(0, 82, 0, 82)
    grid.CellPadding = UDim2.new(0, 7, 0, 7)
    grid.FillDirection = Enum.FillDirection.Horizontal
    grid.SortOrder = Enum.SortOrder.LayoutOrder
    grid.HorizontalAlignment = Enum.HorizontalAlignment.Center
    grid.Parent = gallery

    local selectedInfo = makeLabel(
        currentPage,
        "♡ Tap a sticker to preview it ♡",
        UDim2.new(1, -12, 0, 35),
        UDim2.new(0, 6, 1, -42)
    )
    selectedInfo.Font = Enum.Font.Gotham
    selectedInfo.TextColor3 = Theme.BlueDark

    local function loadCategory(category)
        for _, child in ipairs(gallery:GetChildren()) do
            if child:IsA("ImageButton")
                or child:IsA("TextLabel") then
                child:Destroy()
            end
        end

        gallery.CanvasPosition = Vector2.zero

        local stickers = StickerLibrary[category] or {}

        for index, sticker in ipairs(stickers) do
            local imageButton = Instance.new("ImageButton")
            imageButton.Name = "Sticker_" .. index
            imageButton.LayoutOrder = index
            imageButton.Size = UDim2.fromOffset(82, 82)
            imageButton.BackgroundColor3 = Theme.Pink
            imageButton.BorderSizePixel = 0
            imageButton.Image = sticker.Id
            imageButton.ScaleType = Enum.ScaleType.Fit
            imageButton.Parent = gallery

            makeCorner(imageButton, 12)
            makeStroke(imageButton, Theme.Blue, 1)

            imageButton.Activated:Connect(function()
                selectedInfo.Text = "♡ Selected " .. category .. " sticker #" .. index .. " ♡"

                TweenService:Create(
                    imageButton,
                    TweenInfo.new(0.12),
                    {BackgroundColor3 = Theme.Green}
                ):Play()

                task.delay(0.35, function()
                    if imageButton.Parent then
                        imageButton.BackgroundColor3 = Theme.Pink
                    end
                end)
            end)
        end

        if #stickers == 0 then
            local empty = makeLabel(
                gallery,
                "No stickers here yet ♡",
                UDim2.new(1, -12, 0, 55),
                UDim2.new(0, 6, 0, 10)
            )
            empty.Font = Enum.Font.Gotham
        end
    end

    for _, category in ipairs(Categories) do
        local categoryButton = makeButton(
            categoryBar,
            CategoryNames[category] or category,
            UDim2.fromOffset(112, 36),
            UDim2.new(),
            Theme.Pink
        )

        categoryButton.TextSize = 12

        categoryButton.Activated:Connect(function()
            loadCategory(category)
        end)
    end

    loadCategory("HIKARU")
end

local function showPage(tabName)
    if tabName == "Stickers" then
        showStickers()
        return
    end

    clearPage()

    local pageContent = {
        Home = "Welcome to your little world! ♡\n\n🌸  ✿  🦋  ✧  💗",
        DMs = "Direct Messages ♡\n\nChat integration needs a server setup.",
        Profile = "Your Profile ♡\n\nStay cute, stay you! 🌷",
        Rank = "Your Rank ✨\n\nRank system is a UI placeholder.",
        Settings = "Settings ⚙️\n\n♡ Pastel Blue Theme ♡"
    }

    local message = makeLabel(
        currentPage,
        pageContent[tabName] or tabName,
        UDim2.new(0.9, 0, 0.75, 0),
        UDim2.new(0.05, 0, 0.12, 0)
    )

    message.TextWrapped = true
    message.Font = Enum.Font.GothamBold
end

-- 🌷 LOADING ANIMATION
--==================================================

task.spawn(function()
    for i = 1, 3 do
        loadingText.Text =
            "Loading your little world" .. string.rep(".", i)

        task.wait(0.4)
    end

    local tween = TweenService:Create(
        welcome,
        TweenInfo.new(0.45),
        {BackgroundTransparency = 1}
    )

    tween:Play()

    for _, child in ipairs(welcome:GetDescendants()) do
        if child:IsA("TextLabel") then
            TweenService:Create(
                child,
                TweenInfo.new(0.3),
                {TextTransparency = 1}
            ):Play()
        end
    end

    tween.Completed:Wait()

    welcome:Destroy()
    main.Visible = true

    showPage("Home")
end)
